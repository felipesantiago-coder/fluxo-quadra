    56→// ─── Unit Card (compact grid card) ───
    57→function UnitCard({
    58→  unit,
    59→  onSelect,
    60→  isBackground,
    61→  isAdmin,
    62→  onStatusChange,
    63→}: {
    64→  unit: Unit;
    65→  onSelect: (unit: Unit) => void;
    66→  isBackground: boolean;
    67→  isAdmin?: boolean;
    68→  onStatusChange?: (unidade: number, newStatus: Unit["status"]) => void;
    69→}) {
    70→  const colors = typeColors[unit.tipoArea];
    71→  const status = statusLabels[unit.status];
    72→  const [showStatusMenu, setShowStatusMenu] = useState(false);
    73→  const [saving, setSaving] = useState(false);
    74→  const [feedback, setFeedback] = useState<"success" | "error" | null>(null);
    75→
    76→  // Fechar o menu ao clicar fora
    77→  useEffect(() => {
    78→    if (!showStatusMenu) return;
    79→    const handleClickOutside = () => setShowStatusMenu(false);
    80→    document.addEventListener("click", handleClickOutside);
    81→    return () => document.removeEventListener("click", handleClickOutside);
    82→  }, [showStatusMenu]);
    83→
    84→  // Limpar feedback após 3 segundos
    85→  useEffect(() => {
    86→    if (!feedback) return;
    87→    const timer = setTimeout(() => setFeedback(null), 3000);
    88→    return () => clearTimeout(timer);
    89→  }, [feedback]);
    90→
    91→  const handleStatusClick = (e: React.MouseEvent) => {
    92→    e.stopPropagation();
    93→    if (isAdmin) setShowStatusMenu(!showStatusMenu);
    94→  };
    95→
    96→  const handleStatusSelect = async (e: React.MouseEvent, newStatus: Unit["status"]) => {
    97→    e.stopPropagation();
    98→    setShowStatusMenu(false);
    99→    if (!onStatusChange || newStatus === unit.status) return;
   100→
   101→    setSaving(true);
   102→    setFeedback(null);
   103→    try {
   104→      const res = await fetch("/api/units", {
   105→        method: "PATCH",
   106→        credentials: "include",
   107→        headers: { "Content-Type": "application/json" },
   108→        body: JSON.stringify({ unidade: unit.unidade, status: newStatus }),
   109→      });
   110→      const data = await res.json();
   111→      if (res.ok) {
   112→        onStatusChange(unit.unidade, newStatus);
   113→        setFeedback("success");
   114→      } else {
   115→        console.error("Erro ao atualizar status:", data.error);
   116→        setFeedback("error");
   117→      }
   118→    } catch (err) {
   119→      console.error("Erro ao atualizar status:", err);
   120→      setFeedback("error");
   121→    } finally {
   122→      setSaving(false);
   123→    }
   124→  };
   125→
   126→  return (
   127→    <motion.div
   128→      layout
   129→      initial={{ opacity: 0, y: 20 }}
   130→      animate={{
   131→        opacity: isBackground ? 0.25 : 1,
   132→        y: 0,
   133→      }}
   134→      exit={{ opacity: 0, y: -20 }}
   135→      transition={{
   136→        layout: { type: "spring", stiffness: 300, damping: 30 },
   137→        opacity: { duration: 0.3 },
   138→      }}
   139→      whileHover={!isBackground ? { y: -6, scale: 1.03 } : {}}
   140→      onClick={() => onSelect(unit)}
   141→      className={`
   142→        relative cursor-pointer rounded-xl border-2 overflow-visible
   143→        bg-white shadow-md hover:shadow-xl
   144→        transition-all duration-300 ease-out
   145→        border-gray-100
   146→        ${isBackground ? "pointer-events-none" : ""}
   147→      `}
   148→      style={{
   149→        filter: isBackground ? "blur(2px)" : "none",
   150→      }}
   151→    >
   152→      {/* Top colored bar */}
   153→      <div className={`h-1.5 bg-gradient-to-r ${colors.gradient}`} />
   154→
   155→      <div className="p-5 space-y-3">
   156→        {/* Header: Unit number + Status */}
   157→        <div className="flex items-center justify-between">
   158→          <span className="text-xl font-bold tracking-tight text-gray-900">
   159→            {unit.unidade}
   160→          </span>
   161→          <div className="relative">
   162→            <button
   163→              onClick={handleStatusClick}
   164→              className={`inline-flex items-center gap-1.5 text-[10px] font-semibold uppercase tracking-wider px-2 py-0.5 rounded-full border ${status.color} ${isAdmin ? "cursor-pointer hover:opacity-80 ring-1 ring-offset-1 ring-gray-200 hover:ring-gray-400" : "cursor-default"}`}
   165→            >
   166→              {saving ? (
   167→                <span className="w-1.5 h-1.5 rounded-full bg-gray-400 animate-pulse" />
   168→              ) : (
   169→                <span className={`w-1.5 h-1.5 rounded-full ${status.dotColor}`} />
   170→              )}
   171→              {status.label}
   172→              {isAdmin && !showStatusMenu && <span className="ml-0.5 opacity-50">▾</span>}
   173→            </button>
   174→
   175→            {/* Dropdown de status */}
   176→            <AnimatePresence>
   177→              {showStatusMenu && isAdmin && (
   178→                <motion.div
   179→                  initial={{ opacity: 0, y: -4, scale: 0.95 }}
   180→                  animate={{ opacity: 1, y: 0, scale: 1 }}
   181→                  exit={{ opacity: 0, y: -4, scale: 0.95 }}
   182→                  transition={{ duration: 0.15 }}
   183→                  className="absolute right-0 top-full mt-1 z-50 bg-white rounded-xl shadow-xl border border-gray-200 py-1 min-w-[140px] overflow-hidden"
   184→                  onClick={(e) => e.stopPropagation()}
   185→                >
   186→                  <p className="text-[9px] font-semibold text-gray-400 uppercase tracking-wider px-3 py-1.5">Alterar status</p>
   187→                  {allStatuses.map((s) => (
   188→                    <button
   189→                      key={s.value}
   190→                      onClick={(e) => handleStatusSelect(e, s.value)}
   191→                      className={`w-full flex items-center gap-2 px-3 py-2 text-xs font-medium transition-colors ${
   192→                        s.value === unit.status
   193→                          ? "bg-gray-50 text-gray-400"
   194→                          : "hover:bg-gray-50 text-gray-700"
   195→                      }`}
   196→                    >
   197→                      <span className={`w-2 h-2 rounded-full ${s.dotColor}`} />
   198→                      {s.label}
   199→                      {s.value === unit.status && <Check className="w-3 h-3 ml-auto" />}
   200→                    </button>
   201→                  ))}
   202→                </motion.div>
   203→              )}
   204→            </AnimatePresence>
   205→          </div>
   206→        </div>
   207→
   208→        {/* Feedback visual de sucesso/erro */}
   209→        <AnimatePresence>
   210→          {feedback && (
   211→            <motion.div
   212→              initial={{ opacity: 0, height: 0 }}
   213→              animate={{ opacity: 1, height: "auto" }}
   214→              exit={{ opacity: 0, height: 0 }}
   215→              className={`flex items-center gap-1.5 text-[10px] font-semibold px-2.5 py-1 rounded-lg ${
   216→                feedback === "success"
   217→                  ? "bg-emerald-50 text-emerald-700 border border-emerald-200"
   218→                  : "bg-red-50 text-red-700 border border-red-200"
   219→              }`}
   220→            >
   221→              {feedback === "success" ? (
   222→                <Check className="w-3 h-3 flex-shrink-0" />
   223→              ) : (
   224→                <X className="w-3 h-3 flex-shrink-0" />
   225→              )}
   226→              {feedback === "success" ? "Status atualizado!" : "Erro ao atualizar. Verifique o console."}
   227→            </motion.div>
   228→          )}
   229→        </AnimatePresence>
   230→
   231→        {/* Info items */}
   232→        <div className="grid grid-cols-2 gap-2">
   233→          <div className="flex items-center gap-1.5 text-gray-500">
   234→            <Maximize2 className="w-3.5 h-3.5" />
   235→            <span className="text-sm font-medium">{unit.areaStr}</span>
   236→          </div>
   237→          <div className="flex items-center gap-1.5 text-gray-500">
   238→            <Car className="w-3.5 h-3.5" />
   239→            <span className="text-sm font-medium">{unit.vagas} vag{unit.vagas === 1 ? "a" : "as"}</span>
   240→          </div>
   241→          <div className="flex items-center gap-1.5 text-gray-500">
   242→            <BedDouble className="w-3.5 h-3.5" />
   243→            <span className="text-sm font-medium">{unit.quartos} qts</span>
   244→          </div>
   245→          <div className="flex items-center gap-1.5 text-gray-500">
   246→            <Sun className={`w-3.5 h-3.5 ${unit.posicaoSolar === "Nascente" ? "text-amber-500" : "text-orange-500"}`} />
   247→            <span className="text-sm font-medium">{unit.posicaoSolar}</span>
   248→          </div>
   249→        </div>
   250→
   251→        {/* Price */}
   252→        <div className="pt-1">
   253→          <p className={`text-lg font-bold ${unit.valorVenda ? "text-gray-900" : "text-gray-400 italic"}`}>
   254→            {unit.valorFormatado}
   255→          </p>
   256→          {unit.valorVenda && (
   257→            <p className="text-[11px] text-gray-400 mt-0.5">
   258→              R$ {(unit.valorVenda / unit.area).toLocaleString("pt-BR", { minimumFractionDigits: 0, maximumFractionDigits: 0 })}/m²
   259→            </p>
   260→          )}
   261→        </div>
   262→      </div>
   263→    </motion.div>
   264→  );
   265→}

   424→// ─── Floor Section ───
   425→function FloorSection({
   426→  floor,
   427→  floorUnits,
   428→  selectedUnit,
   429→  onSelectUnit,
   430→  isCollapsed,
   431→  onToggle,
   432→  isAdmin,
   433→  onStatusChange,
   434→}: {
   435→  floor: number;
   436→  floorUnits: Unit[];
   437→  selectedUnit: Unit | null;
   438→  onSelectUnit: (unit: Unit) => void;
   439→  isCollapsed: boolean;
   440→  onToggle: () => void;
   441→  isAdmin?: boolean;
   442→  onStatusChange?: (unidade: number, newStatus: Unit["status"]) => void;
   443→}) {
   444→  return (
   445→    <motion.div layout className="space-y-4">
   446→      {/* Floor header */}
   447→      <motion.button
   448→        onClick={onToggle}
   449→        className="w-full flex items-center justify-between p-4 rounded-xl bg-gradient-to-r from-gray-900 to-gray-800 text-white shadow-lg hover:shadow-xl transition-shadow group"
   450→        whileHover={{ scale: 1.005 }}
   451→        whileTap={{ scale: 0.995 }}
   452→      >
   453→        <div className="flex items-center gap-3">
   454→          <div className="w-10 h-10 rounded-lg bg-white/10 flex items-center justify-center backdrop-blur-sm">
   455→            <Building2 className="w-5 h-5" />
   456→          </div>
   457→          <div className="text-left">
   458→            <h3 className="text-lg font-bold">{floor}º Andar</h3>
   459→            <p className="text-sm text-white/60">{floorUnits.length} unidades</p>
   460→          </div>
   461→        </div>
   462→        <div className="flex items-center gap-3">
   463→          <div className="hidden sm:flex items-center gap-2">
   464→            {areaTypes.map((type) => {
   465→              const count = floorUnits.filter((u) => u.tipoArea === type).length;
   466→              if (count === 0) return null;
   467→              return (
   468→                <Badge key={type} variant="secondary" className="text-[10px] font-semibold bg-white/15 text-white/80 border-white/20">
   469→                  {count}x {type}
   470→                </Badge>
   471→              );
   472→            })}
   473→          </div>
   474→          <motion.div animate={{ rotate: isCollapsed ? 0 : 180 }}>
   475→            <ChevronUp className="w-5 h-5 text-white/60" />
   476→          </motion.div>
   477→        </div>
   478→      </motion.button>
   479→
   480→      {/* Floor units grid */}
   481→      <AnimatePresence>
   482→        {!isCollapsed && (
   483→          <motion.div
   484→            initial={{ height: 0, opacity: 0 }}
   485→            animate={{ height: "auto", opacity: 1 }}
   486→            exit={{ height: 0, opacity: 0 }}

   488→            className="overflow-visible"

   542→        const res = await fetch("/api/units");
   543→        const data = await res.json();

   545→        const mapped: Unit[] = (Array.isArray(data) ? data : []).map((row: Record<string, unknown>) => ({
   546→          andar: row.andar as number,
   547→          unidade: row.unidade as number,
   548→          vagas: row.vagas as number,
   549→          area: Number(row.area),
   550→          areaStr: row.area_str as string,
   551→          valorVenda: row.valor_venda as number | null,
   552→          valorStr: row.valor_venda ? Number(row.valor_venda).toLocaleString("pt-BR", { minimumFractionDigits: 2, maximumFractionDigits: 2 }) : "Consulte",
   553→          valorFormatado: row.valor_venda ? formatCurrency(Number(row.valor_venda)) : "Consulte o valor",
   554→          tipoArea: row.tipo_area as Unit["tipoArea"],
   555→          status: row.status as Unit["status"],
   556→          posicaoSolar: row.posicao_solar as Unit["posicaoSolar"],
   557→          quartos: row.quartos as 2 | 3,
   558→        }));

   565→          .channel("dashboard-realtime")

   567→            "postgres_changes",
   568→            { event: "UPDATE", schema: "public", table: "units" },
   569→            (payload) => {
   570→              const updated = payload.new as Record<string, unknown>;
   571→              setUnits((prev) =>
   572→                prev.map((u) =>
   573→                  u.unidade === updated.unidade
   574→                    ? { ...u, status: updated.status as Unit["status"] }
   575→                    : u
   576→                )
   577→              );
   578→            }
   579→          )
   580→          .subscribe();
   581→      } catch {
   582→        console.error("Erro ao carregar dados do Supabase, usando dados estáticos.");
   583→      }
   584→    }
   585→
   586→    loadData();
   587→
   588→    return () => {
   589→      if (supabaseChannel) {
   590→        createClient().removeChannel(supabaseChannel);
   591→      }
   592→    };
   593→  }, []);
   594→
   595→  const filteredUnits = useMemo(() => {
   596→    let result = [...units];
   597→    if (filterQuartos !== "all") result = result.filter((u) => u.quartos === filterQuartos);
   598→    if (filterSolar !== "all") result = result.filter((u) => u.posicaoSolar === filterSolar);
   599→    if (filterVagas !== "all") result = result.filter((u) => u.vagas === filterVagas);
   600→    if (filterStatus !== "all") result = result.filter((u) => u.status === filterStatus);
   601→    if (sortBy === "price-asc") result.sort((a, b) => (a.valorVenda ?? Infinity) - (b.valorVenda ?? Infinity));
   602→    if (sortBy === "price-desc") result.sort((a, b) => (b.valorVenda ?? 0) - (a.valorVenda ?? 0));
   603→    return result;
   604→  }, [units, filterQuartos, filterSolar, filterVagas, filterStatus, sortBy]);
   605→
   606→  const activeFloors = useMemo(() => {
   607→    const floorSet = new Set(filteredUnits.map((u) => u.andar));
   608→    return floors.filter((f) => floorSet.has(f));
   609→  }, [filteredUnits]);
   610→
   611→  const handleSelectUnit = useCallback((unit: Unit) => {
   612→    setSelectedUnit(unit);
   613→  }, []);
   614→
   615→  const handleLocalStatusChange = useCallback((unidade: number, newStatus: Unit["status"]) => {
   616→    setUnits((prev) => prev.map((u) => u.unidade === unidade ? { ...u, status: newStatus } : u));
   617→    // Também atualizar selectedUnit se for a mesma unidade
   618→    setSelectedUnit((prev) => prev && prev.unidade === unidade ? { ...prev, status: newStatus } : prev);

   647→              <div className="flex items-center gap-3">
   648→                <div className="w-9 h-9 rounded-lg bg-white/10 backdrop-blur-sm border border-white/10 flex items-center justify-center">
   649→                  <Building2 className="w-5 h-5 text-white" />
   650→                </div>
   651→                <div>
   652→                  <h1 className="text-lg font-bold tracking-tight">
   653→                    Espelho de <span className="text-gray-400 font-normal">Vendas</span>
   654→                  </h1>
   655→                  <p className="text-[11px] text-gray-400 font-medium">Quattre Istambul</p>
   656→                </div>
   657→              </div>
   658→              <div className="flex items-center gap-2">
   659→                <a
   660→                  href="/projetos"
   661→                  className="hidden sm:flex items-center gap-1.5 text-xs text-gray-400 hover:text-white transition-colors"
   662→                >
   663→                  <svg className="w-3.5 h-3.5" fill="none" stroke="currentColor" viewBox="0 0 24 24"><path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M15 19l-7-7 7-7" /></svg>
   664→                  Projetos
   665→                </a>

   667→                  <div className="w-1.5 h-1.5 rounded-full bg-emerald-400 animate-pulse" />
   668→                  Atualização em tempo real

   671→                  onClick={handleLogout}
   672→                  className="flex items-center gap-1.5 px-3 py-1.5 rounded-lg bg-red-500/15 hover:bg-red-500/25 text-red-400 text-xs font-semibold transition-colors border border-red-500/20"
   673→                >
